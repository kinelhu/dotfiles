% -----------------------------------------------------------------------------
% Styled table of contents. Invoke \styledtoc anywhere in the Markdown body
% (with `toc: false`) to place the TOC exactly where you want it — e.g. after a
% short preface — with the same styling as the automatic `toc: true` version.
% -----------------------------------------------------------------------------
\newcommand{\styledtoc}{%
  {\hypersetup{linkcolor=AccentPrimary}%
   \setcounter{tocdepth}{2}%
   \setlength{\parskip}{0pt}%
   \begin{spacing}{0.8}\tableofcontents\end{spacing}%
   \vspace{1em}}%
  \newpage}
