% =============================================================================
% house-style.tex — shared PDF/xelatex house style (single source of truth)
% -----------------------------------------------------------------------------
% Owns: body font (Inter), palette, and the \HeaderFont command (Monaspace Neon
% ExtraBold for section/subsection headers, with graceful fallback).
%
% Each document keeps its OWN \titleformat (numbering/hierarchy differ per template)
% and simply uses \HeaderFont as the header font token. Tables are left at the
% pandoc/booktabs default (no house table styling). Pulled in by:
%   - templates/protocol-template.tex      (\input, before \begin{document})
%   - templates/stats-report-template.tex  (\input, before \begin{document})
%   - header.tex                           (include-in-header for bare .qmd/.Rmd)
%
% Requires xelatex. Idempotent + self-contained. Fonts degrade gracefully when
% not installed.
% =============================================================================
\usepackage{fontspec}

% ── Palette ──────────────────────────────────────────────────────────────────
\definecolor{AccentPrimary}{HTML}{27A9BC}
\definecolor{TextPrimary}{HTML}{1A1A1A}
\definecolor{TextSecondary}{HTML}{949494}

% ── Fonts ────────────────────────────────────────────────────────────────────
% Body: Inter if installed (else leave the document default untouched).
\IfFontExistsTF{Inter}{\setmainfont{Inter}}{}
% Headers: Monaspace Neon ExtraBold (neo-grotesque mono) → else Inter SemiBold → else \bfseries.
% Negative tracking tightens the wide monospaced glyphs at heading sizes.
\IfFontExistsTF{Monaspace Neon ExtraBold}{%
  \newfontfamily\HeaderFont{Monaspace Neon ExtraBold}[LetterSpace=-3.0]%
}{%
  \IfFontExistsTF{Inter}%
    {\newfontfamily\HeaderFont{Inter}[UprightFont={Inter SemiBold}]}%
    {\providecommand{\HeaderFont}{\bfseries}}%
}
