% =============================================================================
% house-style.tex — shared PDF/xelatex house style (single source of truth)
% -----------------------------------------------------------------------------
% Owns: body font (Inter), palette, the \HeaderFont command (Monaspace Neon
% ExtraBold for section/subsection headers, with graceful fallback), and the
% tabularray setup used to render zebra-striped tables (see
% filters/table-zebra.lua — that filter rewrites pandoc tables into
% \begin{longtabs}, which is what actually applies AccentPrimary rules +
% AccentLight row banding; this file just makes the package/colors available).
%
% Each document keeps its OWN \titleformat (numbering/hierarchy differ per template)
% and simply uses \HeaderFont as the header font token. Pulled in by:
%   - templates/protocol-template.tex      (\input, before \begin{document})
%   - templates/stats-report-template.tex  (\input, before \begin{document})
%   - header.tex                           (include-in-header for bare .qmd/.Rmd)
%
% Requires xelatex. Idempotent + self-contained. Fonts degrade gracefully when
% not installed.
% =============================================================================
\usepackage{fontspec}
\usepackage{tabularray}
\UseTblrLibrary{booktabs}

% ── Palette ──────────────────────────────────────────────────────────────────
% >>> kinan palette — generated by themes/kinan/sync-palette.R; edit palette.yml, not here
\definecolor{AccentPrimary}{HTML}{27A9BC}
\definecolor{TextPrimary}{HTML}{1A1A1A}
\definecolor{TextSecondary}{HTML}{949494}
\colorlet{AccentLight}{AccentPrimary!8!white}
% <<< kinan palette

% ── Fonts ────────────────────────────────────────────────────────────────────
% Body: Inter if installed (else leave the document default untouched).
% Known and deliberate: Inter's `calt` swaps brackets for cap-height `.case` variants
% next to capitals and lining figures, so "(1.02-1.55)" renders with taller parens. Those
% glyphs carry no Unicode mapping, so pdftotext and copy-paste drop the parentheses while
% the page displays them correctly. RawFeature={-calt} would fix extraction at the cost of
% the cap-height brackets (and of the arrow ligatures). NOT taken: the tables are the
% copy-paste surface, not the prose, and the typography is worth more here (KEH, 2026-09-01).
\IfFontExistsTF{Inter}{\setmainfont{Inter}}{}
% Headers: Monaspace Neon ExtraBold (neo-grotesque mono) → else Inter SemiBold → else \bfseries.
% Negative tracking tightens the wide monospaced glyphs at heading sizes.
\IfFontExistsTF{Monaspace Neon ExtraBold}{%
  \newfontfamily\HeaderFont{Monaspace Neon ExtraBold}[LetterSpace=-3.0]%
}{%
  \IfFontExistsTF{Inter}%
    {\newfontfamily\HeaderFont{Inter}[UprightFont={Inter SemiBold}]}%
    {\providecommand{\HeaderFont}{\bfseries}}%
}
% Running heads: the same family as the section titles, several weights lighter.
% \HeaderFont is ExtraBold with negative tracking, tuned for \Large headings; at
% \small in TextSecondary it reads as a heading that lost its formatting rather
% than as page furniture. Light keeps the family echo while receding, and survives
% a long section name — Neon is monospaced, so width is the binding constraint.
\IfFontExistsTF{Monaspace Neon Light}{%
  \newfontfamily\RunningFont{Monaspace Neon Light}%
}{%
  \IfFontExistsTF{Inter}%
    {\newfontfamily\RunningFont{Inter}}%
    {\providecommand{\RunningFont}{}}%
}
