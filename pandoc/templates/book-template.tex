% Book / essai template — classic literary layout
% Engine: LuaLaTeX (required — luaotfload finds EB Garamond straight out of
% TeX Live without installing anything into Font Book).
% documentclass: book, \frontmatter/\mainmatter, per-chapter endnotes.
%
% Chapter titles are NOT auto-numbered by LaTeX: this template assumes the
% source headings already carry whatever numbering the author wants in the
% text itself (e.g. "## II. Le socle"), so \chapter only ever displays the
% heading text, centered, with no "Chapitre N" label. Unnumbered chapters
% (Introduction, Coda...) just work with no extra markup.

\documentclass[11pt,twoside,openright]{book}

% =============================================================================
% FONTS & LANGUAGE
% =============================================================================
\usepackage{iftex}
\ifnum 0\ifxetex 1\fi\ifluatex 1\fi=0
  \PackageError{book-template}{This template requires XeLaTeX or LuaLaTeX}{Set pdf-engine: lualatex in your defaults file.}
\fi

\usepackage{fontspec}
\usepackage{polyglossia}
\setmainlanguage[]{french}

\setmainfont{EB Garamond}[
  Numbers = OldStyle,
  Ligatures = TeX
]
\newfontfamily\chapterfont{EB Garamond}[Numbers=OldStyle]
\newfontfamily\runningheadfont{EB Garamond}

\usepackage{microtype}

% =============================================================================
% PAGE GEOMETRY
% =============================================================================
$if(geometry)$
\usepackage{geometry}
\geometry{$for(geometry)$$geometry$$sep$,$endfor$}
$else$

$endif$

\definecolor{InkSecondary}{HTML}{6B6B6B}

\renewcommand{\baselinestretch}{1.12}

% =============================================================================
% RUNNING HEADS
% =============================================================================
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0pt}
\renewcommand{\headrule}{\color{RuleGray}\hrule height\headrulewidth}
\fancyhead[LE]{\runningheadfont\small\itshape\color{InkSecondary}\leftmark}
% The book class uppercases \leftmark by default; keep the chapter title as
% typed (it is displayed in italic, uppercase would shout).
\renewcommand{\chaptermark}[1]{\markboth{#1}{}}
\fancyhead[RO]{\runningheadfont\small\itshape\color{InkSecondary}$if(title)$$title$$endif$}
\fancyfoot[LE,RO]{\runningheadfont\small\color{InkSecondary}\thepage}
% Chapter-opening pages use LaTeX's plain-vanilla \pagestyle{plain} (blank
% head, centered footer page number) — the classic book convention. Do NOT
% redefine it via \fancypagestyle{plain}{\fancyhf{}...}: invoking \fancyhf{}
% there clobbers the fancy style's own head/foot storage globally the next
% time a chapter-opening page is typeset, silently blanking every running
% head for the rest of the document.

% openright inserts a blank verso before each chapter when needed; without
% this, that inserted page still carries the running head of the chapter
% just ended. Standard fix: the filler page is typeset truly empty.
\makeatletter
\def\cleardoublepage{\clearpage\if@twoside\ifodd\c@page\else
  \hbox{}\thispagestyle{empty}\newpage
  \if@twocolumn\hbox{}\newpage\fi\fi\fi}
\makeatother

% =============================================================================
% CHAPTER TITLE STYLE — centered, no auto-number, thin rule below
% =============================================================================
\usepackage{titlesec}

% Small caps, gently letterspaced — the classic French essay convention —
% rather than full uppercase, which reads heavy at chapter scale.
\titleformat{\chapter}[display]
  {\centering\chapterfont}
  {}
  {0pt}
  {\Large\scshape\addfontfeature{LetterSpace=6.0}}
  [\vspace{0.4em}{\color{RuleGray}\titlerule[0.4pt]}]

\titlespacing*{\chapter}{0pt}{2.5cm}{1.5cm}

\titleformat{\section}
  {\chapterfont\large\itshape}
  {}
  {0pt}
  {}
\titlespacing*{\section}{0pt}{1.4em}{0.6em}

\titleformat{\subsection}
  {\chapterfont\normalsize\bfseries}
  {}
  {0pt}
  {}
\titlespacing*{\subsection}{0pt}{1.1em}{0.4em}

% Chapter numbers are hand-typed into the heading text by the author (e.g.
% "II. Le socle"), so LaTeX's own chapter counter is never displayed — not
% in the running \chapter title (titleformat above already omits it) and
% not in the table of contents (secnumdepth<0 removes the \numberline too).
\setcounter{secnumdepth}{-1}

% =============================================================================
% OPTIONAL LITERARY DEVICES — available, not auto-applied
% =============================================================================
% Drop cap for a chapter's opening word, invoked manually in the markdown as
% raw LaTeX: \dropcap{I}{l existe un lieu...}
\usepackage{lettrine}
\newcommand{\dropcap}[2]{\lettrine[lines=3,findent=0.1em,nindent=0pt]{#1}{#2}}

% Epigraph, invoked manually as raw LaTeX right after a chapter heading:
% \epigraph{quotation}{--- attribution}
\usepackage{epigraph}
\setlength{\epigraphwidth}{0.7\textwidth}
\renewcommand{\epigraphflush}{flushright}
\renewcommand{\epigraphrule}{0pt}
\renewcommand{\textflush}{flushright}

% =============================================================================
% PER-CHAPTER ENDNOTES
% =============================================================================
% Markdown footnotes ([^1]) become \endnote instead of \footnote, and are
% flushed automatically right before each new \chapter starts (and once more
% at the very end, via \flushendnotes before \end{document}).
\usepackage{endnotes}
\let\footnote=\endnote
\renewcommand{\notesname}{Notes}
\renewcommand{\enotesize}{\small}
\definecolor{EndnoteRule}{HTML}{B0B0B0}
\renewcommand{\enoteheading}{%
  \vspace{1.5em}
  {\color{EndnoteRule}\titlerule[0.4pt]}
  \vspace{0.6em}
  \noindent{\chapterfont\itshape\notesname}\par
  \mbox{}\par\vskip-\baselineskip
}

\usepackage{etoolbox}
\pretocmd{\chapter}{%
  \ifnum\value{endnote}>0
    {\theendnotes}
    \setcounter{endnote}{0}
  \fi
}{}{}
\newcommand{\flushendnotes}{%
  \ifnum\value{endnote}>0
    {\theendnotes}
    \setcounter{endnote}{0}
  \fi
}
